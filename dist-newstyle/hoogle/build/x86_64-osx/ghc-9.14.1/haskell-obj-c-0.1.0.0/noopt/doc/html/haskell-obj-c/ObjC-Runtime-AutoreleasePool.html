objc_autoreleasePoolPush :: IO (Ptr ()) Source #

Push a new autorelease pool. Returns a token to pass to objc_autoreleasePoolPop.

objc_autoreleasePoolPop :: Ptr () -> IO () Source #

Pop an autorelease pool (releasing all autoreleased objects since the corresponding push).

withAutoreleasePool :: IO a -> IO a Source #

Execute an action inside an autorelease pool. The pool is drained after the action completes (or on exception).
